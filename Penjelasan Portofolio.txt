'use client';
import { useState, useEffect } from 'react';
import { usePathname } from 'next/navigation';
import Navbar from './navbar';
import Sidebar from './sidebar';

const Layout = ({ children }) => {
    const pathname = usePathname();
    const [activeSection, setActiveSection] = useState('hero');
    const [isClient, setIsClient] = useState(false);

    // Check if we're on homepage
    const isHomePage = pathname === '/';

    // Handle hydration
    useEffect(() => {
        setIsClient(true);
    }, []);

    const handleSectionChange = (sectionId) => {
        setActiveSection(sectionId);
    };

    // Enhanced Intersection Observer for automatic section detection (only on homepage)
    useEffect(() => {
        if (!isHomePage || !isClient) return;

        const observerOptions = {
            root: null,
            rootMargin: '-40% 0px -40% 0px',
            threshold: [0, 0.1, 0.5, 0.9, 1]
        };

        const observerCallback = (entries) => {
            let bestEntry = null;
            let highestRatio = 0;

            entries.forEach((entry) => {
                if (entry.isIntersecting && entry.intersectionRatio > highestRatio) {
                    highestRatio = entry.intersectionRatio;
                    bestEntry = entry;
                }
            });

            if (bestEntry) {
                setActiveSection(bestEntry.target.id);
            }
        };

        const observer = new IntersectionObserver(observerCallback, observerOptions);

        const sections = [
            'hero',
            'about-preview',
            'projects-preview',
            'skills-preview',
            'contact-preview'
        ];

        sections.forEach((sectionId) => {
            const element = document.getElementById(sectionId);
            if (element) {
                observer.observe(element);
            }
        });

        return () => {
            sections.forEach((sectionId) => {
                const element = document.getElementById(sectionId);
                if (element) {
                    observer.unobserve(element);
                }
            });
        };
    }, [isHomePage, isClient]);

    return (
        <div className="min-h-screen bg-gradient-to-br from-white via-gray-400 to-neutral-900">
            {/* Navigation */}  
            <Navbar />

            {/* Layout Container - Conditional Grid based on homepage */}
            <div className={`min-h-[calc(100vh-64px)] ${isHomePage ? 'md:grid md:grid-cols-[200px_1fr]' : ''}`}>
                {/* Desktop/Tablet Sidebar - Only show on homepage after client hydration */}
                {isHomePage && isClient && (
                    <div className="hidden md:block">
                        <Sidebar
                            activeSection={activeSection}
                            onSectionChange={handleSectionChange}
                        />
                    </div>
                )}

                {/* Main Content Area */}
                <main className={`flex-1 ${isHomePage ? 'md:overflow-x-hidden' : ''}`}>
                    {children}
                </main>
            </div>
        </div>
    );
};

import { 
    Eye, Download, Send, ArrowRight, ExternalLink, 
    Github, Linkedin, Mail, Phone, Calendar,
    Loader2, Check
} from 'lucide-react';

// Main Button Component
export const Button = ({ 
    variant = 'primary', 
    size = 'md', 
    children, 
    icon: Icon,
    iconPosition = 'left',
    disabled = false,
    loading = false,
    onClick,
    href,
    target,
    className = '',
    ...props 
}) => {
    // Base styles
    const baseStyles = "inline-flex items-center justify-center font-semibold transition-all duration-300 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-offset-slate-900";
    
    // Variant styles
    const variants = {
        primary: "bg-blue-600 hover:bg-blue-700 text-white shadow-lg hover:shadow-xl hover:shadow-blue-900/20 focus:ring-blue-500 hover:scale-105",
        secondary: "bg-slate-800/50 backdrop-blur-sm border-2 border-slate-700 text-white hover:bg-slate-700/50 hover:border-slate-600 focus:ring-slate-500 hover:scale-105",
        outline: "border-2 border-blue-600 text-blue-600 hover:bg-blue-600 hover:text-white focus:ring-blue-500 hover:scale-105",
        ghost: "text-slate-300 hover:text-white hover:bg-slate-800/30",
        danger: "bg-red-600 hover:bg-red-700 text-white shadow-lg hover:shadow-xl hover:shadow-red-900/20 focus:ring-red-500 hover:scale-105",
        success: "bg-green-600 hover:bg-green-700 text-white shadow-lg hover:shadow-xl hover:shadow-green-900/20 focus:ring-green-500 hover:scale-105"
    };
    
    // Size styles
    const sizes = {
        sm: "px-4 py-2 text-sm rounded-md gap-2",
        md: "px-6 py-3 text-base rounded-2xl gap-3",
        lg: "px-8 py-4 text-lg rounded-lg gap-3",
        xl: "px-10 py-5 text-xl rounded-xl gap-4"
    };
    
    // Disabled styles
    const disabledStyles = disabled || loading 
        ? "opacity-50 cursor-not-allowed hover:scale-100" 
        : "";
    
    const combinedClassName = `${baseStyles} ${variants[variant]} ${sizes[size]} ${disabledStyles} ${className}`;
    
    const content = (
        <>
            {loading && <Loader2 className="w-4 h-4 animate-spin" />}
            {!loading && Icon && iconPosition === 'left' && <Icon className="w-5 h-5 group-hover:rotate-12 transition-transform duration-300" />}
            {children}
            {!loading && Icon && iconPosition === 'right' && <Icon className="w-5 h-5 group-hover:translate-x-1 transition-transform duration-300" />}
        </>
    );
    
    if (href) {
        return (
            <a
                href={href}
                target={target}
                className={`${combinedClassName} group`}
                {...props}
            >
                {content}
            </a>
        );
    }
    
    return (
        <button
            onClick={onClick}
            disabled={disabled || loading}
            className={`${combinedClassName} group`}
            {...props}
        >
            {content}
        </button>
    );
};
